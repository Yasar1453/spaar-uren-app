 E. Aanroepen vanuit de bestaande modules
----------------------------------------------------------------------------
 communicatie.js maakt zelf geen Supabase-client en dupliceert geen
 hulpfuncties: allebei komen ze mee als argument. `mijnId` is een functie en
 geen waarde, omdat het medewerker-id pas ná het inloggen bekend is.
============================================================================

--- beheer.js -------------------------------------------------------------

  // bovenaan, bij de andere imports
  import { bouwNieuwsBeheer } from "./communicatie.js?v=42";

  // in naarDash(), ná het bepalen van ikBenId en ná laadBeleid()/laadVormen()
  const communicatie = bouwNieuwsBeheer(db, {
    esc, rijLeeg, toonMeld, datum, tijd, haalAlles, initialen,
    mijnId: () => ikBenId,
  });
  await communicatie.laad();

--- monteur.js ------------------------------------------------------------

  // bovenaan, bij de andere imports
  import { bouwNieuwsMonteur } from "./communicatie.js?v=42";

  // in naarStatus(), ná laadHome()  —  let op: monteur.js noemt zijn
  // meldingfunctie toonMelding; de module verwacht hem als toonMeld.
  const communicatie = bouwNieuwsMonteur(db, {
    esc, netfout, toonMeld: toonMelding,
    mijnId: () => (mij ? mij.medewerker_id : null),
  });
  communicatie.laad();
